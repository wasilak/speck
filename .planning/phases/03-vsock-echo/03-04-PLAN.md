    **Create integration test file:**

    ```rust
    //! End-to-end test: boot a micro-VM with vminitd, connect via vsock, echo round-trip.
    //!
    //! This test requires:
    //!   - A Kata arm64 kernel at the path specified by KERNEL env var or default
    //!   - A custom initrd at the path specified by INITRD env var or default
    //!   - The host to have the com.apple.security.virtualization entitlement
    //!
    //! Run with: cargo test -p speck-vz --test vsock_echo_integration_test -- --ignored
    //! Or: cargo test -p speck-vz --test vsock_echo_integration_test -- --include-ignored
    //!
    //! The test is #[ignore]d by default because it requires VM-entitled execution.

    use speck_vz::{Guest, GuestConfig, Error};
    use std::path::PathBuf;
    use std::time::Duration;

    /// Paths to boot artifacts — override with env vars.
    fn kernel_path() -> PathBuf {
        std::env::var("SPEK_KERNEL")
            .unwrap_or_else(|_| {
                let base = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
                base.join("../../target/kernel/vmlinux")
            })
    }

    fn initrd_path() -> PathBuf {
        std::env::var("SPEK_INITRD")
            .unwrap_or_else(|_| "/tmp/speck-initrd.cpio.gz".into())
    }

    #[test]
    #[ignore = "requires VM entitlement + kernel + initrd artifacts"]
    fn test_vsock_echo() {
        // 1. Build guest config
        let config = GuestConfig::builder()
            .kernel_path(kernel_path())
            .initrd_path(Some(initrd_path()))
            .cmdline("console=hvc0 vsock_port=1234".to_string())
            .cpu_count(1)
            .memory_size_bytes(512 * 1024 * 1024) // 512 MB
            .vsock_port(1234)
            .stop_timeout(Duration::from_secs(10))
            .build();

        // 2. Create and start the guest
        let mut guest = Guest::new().expect("Guest::new");
        guest.start(&config).expect("guest start");

        // 3. Wait for the guest to boot and vminitd to listen
        //    The Kata kernel + minimal initrd boots in < 1s.
        //    2s is conservative.
        std::thread::sleep(Duration::from_secs(2));

        // 4. Connect via vsock
        let mut sock = guest.vsock_connect(1234).expect("vsock_connect");

        // 5. Write test payload
        let payload = b"hello speck";
        let n_written = sock.write(payload).expect("write");
        assert_eq!(n_written, payload.len(), "wrote entire payload");

        // 6. Read back echo
        let mut buf = [0u8; 4096];
        let n_read = sock.read(&mut buf).expect("read");
        let echo = &buf[..n_read];

        assert_eq!(echo, payload, "echo matches sent payload");

        // 7. Clean up — drop sock closes fd, stop destroys the VM
        drop(sock);
        guest.stop().expect("guest stop");
    }

    #[test]
    #[ignore = "requires VM entitlement"]
    fn test_vsock_connect_refused() {
        // Verify that connecting to a port with no listener returns VsockTimeout
        let config = GuestConfig::builder()
            .kernel_path(kernel_path())
            .initrd_path(Some(initrd_path()))
            .cmdline("console=hvc0 vsock_port=1234".to_string())
            .cpu_count(1)
            .memory_size_bytes(512 * 1024 * 1024)
            .vsock_port(1234)   // vminitd listens on 1234, not 9999
            .stop_timeout(Duration::from_secs(10))
            .build();

        let mut guest = Guest::new().expect("Guest::new");
        guest.start(&config).expect("guest start");
        std::thread::sleep(Duration::from_secs(2));

        let result = guest.vsock_connect(9999);
        assert!(matches!(result, Err(Error::VsockTimeout)),
            "expected VsockTimeout, got {:?}", result);

        guest.stop().expect("guest stop");
    }
    ```
