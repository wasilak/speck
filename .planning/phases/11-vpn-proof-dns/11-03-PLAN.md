    IMPORTS:
      use std::collections::HashMap;
      use std::net::IpAddr;
      use std::sync::Arc;
      use tokio::sync::watch;
      use tracing::{debug, warn};
      use crate::error;

      use core_foundation::string::CFString;
      Plus system_configuration_sys imports for SCDynamicStoreSetDispatchQueue.

    STRUCT ResolverTable:
      #[derive(Debug, Clone, Default)]
      pub struct ResolverTable {
          entries: HashMap&lt;String, Vec&lt;IpAddr&gt;&gt;,
      }

    IMPL ResolverTable:
      pub fn add_entry(&amp;mut self, domain: String, servers: Vec&lt;IpAddr&gt;) — inserts only when servers is non-empty.
      pub fn find_resolver(&amp;self, domain: &amp;str) -&gt; Option&lt;&amp;[IpAddr]&gt; — longest-suffix match:
        split domain by '.'; iterate from start=0 to len-1; join labels[start..] as suffix; check self.entries.get(&amp;suffix); return first match. Return None if no match.

    PUBLIC FUNCTION spawn_resolver_watcher():
      Signature: pub fn spawn_resolver_watcher() -&gt; (Arc&lt;watch::Sender&lt;ResolverTable&gt;&gt;, watch::Receiver&lt;ResolverTable&gt;)
      Body:
        1. Create channel: let (tx, rx) = watch::channel(ResolverTable::default());
        2. Wrap sender: let tx = Arc::new(tx);
        3. Build a SCDynamicStore using SCDynamicStoreBuilder (high-level crate):
             let callback_tx = Arc::clone(&amp;tx);
             let store = SCDynamicStoreBuilder::new("speck-dns-watcher")
                 .callback(move |store, _changed_keys| {
                     let table = read_resolver_table(&amp;store);
                     debug!(entries = table.entries.len(), "resolver table updated");
                     let _ = callback_tx.send(table);
                 })
                 .build();
           Note: check docs.rs/system-configuration/0.7.0 for the exact Builder API. If
           SCDynamicStoreBuilder does not accept a Rust closure directly, use
           system_configuration_sys::dynamic_store::SCDynamicStoreCreate with an unsafe extern "C"
           callback. Either is correct — prefer the high-level Builder if it compiles.
        4. Set notification keys via store.set_notification_keys with patterns:
             ["State:/Network/Service/[^/]+/DNS", "State:/Network/Global/DNS"]
           Use CFArray of CFString patterns.
        5. Set GCD dispatch queue — THIS IS THE CRITICAL STEP (STATE.md D-09):
             Create a serial GCD queue: let queue = dispatch2::Queue::serial_with_autoreleasepool("speck.dns.watcher");
             Call SCDynamicStoreSetDispatchQueue from system_configuration_sys:
             unsafe { system_configuration_sys::dynamic_store::SCDynamicStoreSetDispatchQueue(store.as_concrete_TypeRef(), queue.as_ptr()) }
             or equivalent. Consult docs.rs/system-configuration-sys/0.6.0 for the exact fn path.
           NEVER use CFRunLoopAddSource — dispatch queue is the only correct model per STATE.md.
        6. Read initial table synchronously before returning:
             let initial = read_resolver_table(&amp;store);
             let _ = tx.send(initial);
           (store must still be live here — do not drop it before this call)
        7. Leak the store and queue into statics or a Box::leak so they are never dropped
           (they must stay alive for the lifetime of the process to keep notifications firing).
        8. Return (tx, rx)

    PRIVATE FUNCTION read_resolver_table(store: &amp;SCDynamicStore) -&gt; ResolverTable:
      1. Use store.get_keys_matching_pattern with pattern "State:/Network/Service/.*/DNS" (CFString) to get all per-service DNS keys.
      2. For each key, call store.get(key) to get the CFDictionary value.
      3. Extract "SupplementalMatchDomains" from the dictionary as CFArray of CFString. If the key is missing or the array is empty, SKIP this service (it is a default resolver, not a VPN-scoped one).
      4. Extract "ServerAddresses" from the dictionary as CFArray of CFString. Parse each string as IpAddr via str::parse::&lt;IpAddr&gt;. Skip entries that fail to parse.
      5. For each domain in SupplementalMatchDomains, call table.add_entry(domain, servers.clone()).
      6. Return the populated table.
      Log the number of VPN-scoped entries found: debug!(count = table.entries.len(), "read resolver table from SCDynamicStore");
      Never call eprintln! or println! — use tracing only.

    ERROR HANDLING: If any SCDynamicStore call returns an error, log with warn! and return an empty table (never panic). Use crate::error::Error::DynamicStore("...".to_string()) where appropriate.

    NO #[allow(unused)] attributes. All functions must be used by the module or exported.
