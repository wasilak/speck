        fn direct_dns_query(nameserver: std::net::IpAddr, query: &amp;[u8]) -&gt; Option&lt;Vec&lt;u8&gt;&gt; {
            use std::net::{SocketAddr, UdpSocket};
            use std::time::Duration;
            let socket = UdpSocket::bind("0.0.0.0:0").ok()?;
            socket.set_read_timeout(Some(Duration::from_millis(500))).ok()?;
            let ns_addr = SocketAddr::new(nameserver, 53);
            socket.send_to(query, ns_addr).ok()?;
            let mut resp_buf = vec![0u8; 4096];
            match socket.recv_from(&amp;mut resp_buf) {
                Ok((n, _)) =&gt; Some(resp_buf[..n].to_vec()),
                Err(_) =&gt; None,
            }
        }

        pub(crate) fn translate_nxdomain_to_servfail(response: &amp;mut [u8]) {
            if response.len() &lt; 4 { return; }
            if response[3] &amp; 0x0f == 3 {
                response[3] = (response[3] &amp; 0xf0) | 0x02;
            }
        }

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| guest UDP :53 / host vsock | DNS queries cross vsock from Linux guest; host parses wire format |
| host / VPN nameserver UDP | Direct UDP to VPN nameserver:53; response bytes parsed for RCODE |
| ResolverTable / dns proxy | GCD thread writes ResolverTable via watch::Sender; proxy reads via borrow_and_update |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-11-04-S1 | Spoofing | DNS response from VPN nameserver (direct UDP) | mitigate | Target known IpAddr from SCDynamicStore ServerAddresses (not hostnames); 500ms read timeout; no caching of VPN responses; SERVFAIL translation prevents client caching on NXDOMAIN |
| T-11-04-T1 | Tampering | Malformed DNS query (oversized or truncated) | mitigate | query_len > buf.len() guard (buf now 4096, up from 512); existing guard at dns.rs line 35 covers oversized; read_exact_fd returns false on short read |
| T-11-04-S2 | Spoofing | NXDOMAIN cache poisoning — VPN resolver returns NXDOMAIN for a name it will later serve | mitigate | DNS-06 SERVFAIL translation prevents client from caching the negative answer; client retries after VPN reconnect |
| T-11-04-I1 | Information Disclosure | DNS query domain names logged via tracing::debug! | accept | Debug logging is disabled in production builds by default; domain names in DNS queries are not sensitive in this threat model |
| T-11-04-SC | Tampering | cargo packages (verified in 11-02) | n/a | No new packages in this plan |
</threat_model>
