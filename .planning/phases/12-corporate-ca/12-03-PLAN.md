    **A. Add `parse_ca_certs_tag` function (after `mount_identity_roots` at line 912):**

    Following the same pattern as the other cmdline parsers:
    ```rust
    /// Parse `ca_certs_tag=TAG` from the kernel cmdline.
    ///
    /// Returns `Some(tag)` if the kernel cmdline includes ca_certs_tag=,
    /// or `None` if no CA cert VirtioFS share is configured (D-12-01).
    fn parse_ca_certs_tag(path: &str) -> Option<String> {
        let content = std::fs::read_to_string(path).ok()?;
        for word in content.split_whitespace() {
            if let Some(val) = word.strip_prefix("ca_certs_tag=") {
                if !val.is_empty() {
                    return Some(val.to_string());
                }
            }
        }
        None
    }
    ```

    **B. Add `mount_ca_certs` function (after `parse_ca_certs_tag`):**

    Following the same `libc::mount` pattern as `mount_identity_roots`:
    ```rust
    /// Mount the CA certificate VirtioFS share.
    ///
    /// Parses `ca_certs_tag` from the kernel cmdline and mounts the
    /// VirtioFS filesystem at `/var/lib/speck/ca-certs/` inside the
    /// guest namespace.  The host populates this directory with validated
    /// PEM files from `$SPECK_HOME/ca-certs/` (CERT-02).
    ///
    /// Non-fatal on error — containers will simply not have custom CA
    /// trust if the mount fails.
    fn mount_ca_certs(cmdline_path: &str) {
        let tag = match parse_ca_certs_tag(cmdline_path) {
            Some(t) => t,
            None => return,
        };
        let target = "/var/lib/speck/ca-certs";
        if let Err(e) = std::fs::create_dir_all(target) {
            eprintln!("vminitd: create_dir_all {target} failed: {e}");
            return;
        }

        let tag_c = std::ffi::CString::new(tag.as_str()).unwrap_or_default();
        let target_c = std::ffi::CString::new(target).unwrap_or_default();

        let ret = unsafe {
            libc::mount(
                tag_c.as_ptr(),
                target_c.as_ptr(),
                b"virtiofs\0".as_ptr() as *const libc::c_char,
                0,
                std::ptr::null(),
            )
        };
        if ret < 0 {
            eprintln!(
                "vminitd: failed to mount CA certs share {tag} at {target}: {:?}",
                io::Error::last_os_error()
            );
        } else {
            eprintln!("vminitd: mounted CA certs share at {target}");
        }
    }
    ```

    **C. Add `install_ca_certs` function (after `mount_ca_certs`):**

    Copies PEMs from `/var/lib/speck/ca-certs/` to `/rootfs/usr/local/share/ca-certificates/` and runs `update-ca-certificates` inside the chroot per D-12-03:
    ```rust
    /// Copy CA certs into the chroot and run update-ca-certificates.
    ///
    /// This must happen AFTER the CA VirtioFS is mounted (the raw PEMs
    /// are at `/var/lib/speck/ca-certs/`) and BEFORE dockerd starts
    /// (dockerd reads the system trust bundle on init).
    ///
    /// Per D-12-03: Alpine uses `/usr/local/share/ca-certificates/`
    /// as input; `update-ca-certificates` produces
    /// `/etc/ssl/certs/ca-certificates.crt`.  Containerd reads the
    /// same bundle via `_default` hosts.toml (D-12-05).
    ///
    /// Non-fatal on error — containers operate without custom CA trust.
    fn install_ca_certs() {
        let ca_dir = "/var/lib/speck/ca-certs";
        let chroot_ca_dir = "/usr/local/share/ca-certificates";
        let chroot_target = "/rootfs/usr/local/share/ca-certificates";

        let entries = match std::fs::read_dir(ca_dir) {
            Ok(e) => e,
            Err(_) => return, // no CA certs configured or mount failed
        };

        // Ensure the target directory exists inside the chroot
        let _ = std::fs::create_dir_all(chroot_target);

        let mut count = 0u32;
        for entry in entries.flatten() {
            let src = entry.path();
            if !src.is_file() {
                continue;
            }
            // Use the same filename (SHA256.pem from the host copy)
            let fname = match src.file_name() {
                Some(n) => n,
                None => continue,
            };
            let dst = std::path::Path::new(chroot_target).join(fname);
            if let Err(e) = std::fs::copy(&src, &dst) {
                eprintln!("vminitd: copy CA cert {src:?} -> {dst:?} failed: {e}");
                continue;
            }
            eprintln!("vminitd: installed CA cert {fname:?}");
            count += 1;
        }

        if count == 0 {
            return;
        }

        // Run update-ca-certificates inside the chroot
        let status = std::process::Command::new("chroot")
            .args(["/rootfs", "update-ca-certificates", "--fresh"])
            .status();
        match status {
            Ok(s) if s.success() => {
                eprintln!("vminitd: update-ca-certificates succeeded ({count} certs)");
            }
            Ok(s) => {
                eprintln!(
                    "vminitd: update-ca-certificates exited with code {:?}",
                    s.code()
                );
            }
            Err(e) => {
                eprintln!("vminitd: update-ca-certificates exec failed: {e}");
            }
        }
    }
    ```

    ```rust
    /// Write containerd `hosts.toml` for CA cert trust (CERT-03).
    ///
    /// Creates `/etc/containerd/certs.d/_default/hosts.toml` that points
    /// containerd at the system trust bundle so pulled images from private
    /// registries with custom CAs verify successfully.
    ///
    /// Per D-12-05: use `_default` catch-all, not per-registry entries.
    /// BuildKit uses the system trust store directly — no BuildKit config
    /// change needed.
    ///
    /// Non-fatal on error — image pulls fall back to host OS trust.
    fn write_containerd_hosts_toml() {
        let dir = "/rootfs/etc/containerd/certs.d/_default";
        if let Err(e) = std::fs::create_dir_all(dir) {
            eprintln!("vminitd: create_dir_all {dir} failed: {e}");
            return;
        }
        let path = std::path::Path::new(dir).join("hosts.toml");
        // Only write if the file doesn't exist yet (don't overwrite user config)
        if path.exists() {
            return;
        }
        // Point containerd at the system trust store that
        // update-ca-certificates produced
        let content = r#"# Written by vminitd for Corporate CA trust (CERT-03)
# Custom CA certificates from config.yaml ca.extra_certs
# are injected into the system trust bundle on every spk up.
server = ""
[host."https://registry-1.docker.io"]
  ca = "/etc/ssl/certs/ca-certificates.crt"
"#;
        if let Err(e) = std::fs::write(&path, content) {
            eprintln!("vminitd: write hosts.toml failed: {e}");
        } else {
            eprintln!("vminitd: wrote containerd hosts.toml at {dir}/hosts.toml");
        }
    }
    ```

|----------|-------------|

| vminitd → /rootfs/etc/containerd/certs.d/ | Config file written into chroot filesystem |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-12-04 | Tampering | VirtioFS CA share | mitigate | Share is read-only (`read_only: true` in VZSharedDirectory); VM has no write path back to host |
| T-12-05 | Tampering | install_ca_certs | mitigate | Only copies PEMs from the read-only VirtioFS mount; no arbitrary file writes (the source is the host-controlled ca-certs/ share) |
| T-12-06 | Tampering | write_containerd_hosts_toml | mitigate | Only writes the _default hosts.toml; does NOT touch existing per-registry files. Uses fixed content without interpolation |
| T-12-07 | Elevation of Privilege | update-ca-certificates in chroot | accept | Runs inside the chroot as root; this is the same privilege level as dockerd itself — no additional attack surface |
| T-12-08 | Info Disclosure | ca-certs/ directory content | accept | Directory lives under $SPECK_HOME which already contains sensitive VM assets; same protection domain |
