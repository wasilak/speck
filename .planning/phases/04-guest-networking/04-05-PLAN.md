//! Integration test: boot a micro-VM with network device, exercise the netstack.
//!
//! This test:
//! 1. Creates GuestConfig with NetworkConfig (subnet, MTU, MAC)
//! 2. Starts the VM (which wires VZFileHandleNetworkDeviceConfiguration)
//! 3. Retrieves the host-side netstack fd
//! 4. Creates FdDevice + SmoltcpInterface from the host fd
//! 5. Sends a synthetic ARP request for the guest IP
//! 6. Polls the interface — verifies the netstack responds with a valid ARP reply
//! 7. Cleans up: stops the VM, closes fd
//!
//! This test requires:
//!   - A Kata arm64 kernel at the path specified by SPEK_KERNEL env var or default
//!   - A custom initrd at the path specified by SPEK_INITRD env var or default
//!   - The host binary to have the com.apple.security.virtualization entitlement
//!
//! Run with: cargo test -p speck-vz --test networking_integration_test -- --ignored
//!
//! The test is #[ignore]d by default because it requires VM-entitled execution.

use speck_vz::{Guest, GuestConfig, Error};
use std::io::{Read, Write};
use std::os::unix::io::RawFd;
use std::path::PathBuf;
use std::time::Duration;

/// Paths to boot artifacts — override with env vars.
fn kernel_path() -> PathBuf {
    std::env::var("SPEK_KERNEL").unwrap_or_else(|_| {
        let base = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
        base.join("../../target/kernel/vmlinux")
    })
}

fn initrd_path() -> PathBuf {
    std::env::var("SPEK_INITRD")
        .unwrap_or_else(|_| "/tmp/speck-initrd.cpio.gz".into())
}

/// Build a minimal valid ARP request targeting the netstack.
///
/// ARP request: who-has 172.16.0.1? tell 172.16.0.2 (guest MAC)
/// Wraps in Ethernet II frame.
fn build_arp_request() -> Vec<u8> {
    let mut frame = Vec::with_capacity(42);

    // Ethernet header
    // Destination MAC: netstack MAC (02:00:00:00:00:01)
    frame.extend_from_slice(&[0x02, 0x00, 0x00, 0x00, 0x00, 0x01]);
    // Source MAC: guest MAC (02:00:00:00:00:02)
    frame.extend_from_slice(&[0x02, 0x00, 0x00, 0x00, 0x00, 0x02]);
    // EtherType: ARP (0x0806)
    frame.extend_from_slice(&[0x08, 0x06]);

    // ARP header (28 bytes)
    // HTYPE: Ethernet (1)
    frame.extend_from_slice(&[0x00, 0x01]);
    // PTYPE: IPv4 (0x0800)
    frame.extend_from_slice(&[0x08, 0x00]);
    // HLEN: 6
    frame.extend_from_slice(&[0x06]);
    // PLEN: 4
    frame.extend_from_slice(&[0x04]);
    // OPER: Request (1)
    frame.extend_from_slice(&[0x00, 0x01]);

    // Sender MAC: 02:00:00:00:00:02 (guest)
    frame.extend_from_slice(&[0x02, 0x00, 0x00, 0x00, 0x00, 0x02]);
    // Sender IP: 172.16.0.2 (guest)
    frame.extend_from_slice(&[172, 16, 0, 2]);
    // Target MAC: 00:00:00:00:00:00 (unknown)
    frame.extend_from_slice(&[0x00, 0x00, 0x00, 0x00, 0x00, 0x00]);
    // Target IP: 172.16.0.1 (netstack/gateway)
    frame.extend_from_slice(&[172, 16, 0, 1]);

    frame
}

/// Parse an Ethernet frame and check if it's an ARP reply.
///
/// Returns true if the frame is a valid ARP reply from 172.16.0.1 to 172.16.0.2.
fn is_arp_reply(frame: &[u8]) -> bool {
    if frame.len() < 42 {
        return false;
    }

    // Check EtherType = ARP (0x0806)
    if frame[12] != 0x08 || frame[13] != 0x06 {
        return false;
    }

    // Check OPER = Reply (2) at offset 20-21 (14 Ethernet + 6 ARP)
    if frame[20] != 0x00 || frame[21] != 0x02 {
        return false;
    }

    // Check sender IP = 172.16.0.1 (netstack) at offset 28-31
    // (14 Ethernet + 14 ARP header before sender IP)
    if frame[28] != 172 || frame[29] != 16 || frame[30] != 0 || frame[31] != 1 {
        return false;
    }

    // Check target IP = 172.16.0.2 (guest) at offset 38-41
    // (14 Ethernet + 24 ARP before target IP)
    if frame[38] != 172 || frame[39] != 16 || frame[40] != 0 || frame[41] != 2 {
        return false;
    }

    true
}

#[test]
#[ignore = "requires VM entitlement + kernel + initrd artifacts"]
fn test_netstack_arp() {
    // 1. Build guest config with networking
    let config = GuestConfig::builder()
        .kernel_path(kernel_path())
        .initrd_path(Some(initrd_path()))
        .cmdline("console=hvc0".to_string())
        .cpu_count(1)
        .memory_size_bytes(512 * 1024 * 1024) // 512 MB
        .vsock_port(1234)
        .network(speck_vz::NetworkConfig {
            subnet: std::net::Ipv4Addr::new(172, 16, 0, 0),
            gateway: std::net::Ipv4Addr::new(172, 16, 0, 1),
            guest_ip: std::net::Ipv4Addr::new(172, 16, 0, 2),
            netmask: 24,
            mtu: 1500,
            mac_address: [0x02, 0x00, 0x00, 0x00, 0x00, 0x02],
        })
        .stop_timeout(Duration::from_secs(10))
        .build();

    // 2. Create and start the guest
    let mut guest = Guest::new().expect("Guest::new");
    guest.start(&config).expect("guest start");

    // 3. Retrieve the host netstack fd
    let host_fd: RawFd = guest.netstack_fd()
        .expect("netstack_fd");

    // 4. Wait briefly for the guest to boot and virtio-net to negotiate
    std::thread::sleep(Duration::from_secs(1));

    // 5. Create FdDevice + SmoltcpInterface
    let device = speck_net::FdDevice::new(host_fd, 1500);
    let mut net = speck_net::SmoltcpInterface::new(device, &speck_net::NetworkConfig {
        subnet: std::net::Ipv4Addr::new(172, 16, 0, 0),
        gateway: std::net::Ipv4Addr::new(172, 16, 0, 1),
        guest_ip: std::net::Ipv4Addr::new(172, 16, 0, 2),
        netmask: 24,
        mtu: 1500,
        mac_address: [0x02, 0x00, 0x00, 0x00, 0x00, 0x01],
    });

    // 6. Send synthetic ARP request
    let arp_request = build_arp_request();
    let fd = host_fd; // raw fd for write
    unsafe {
        libc::write(fd, arp_request.as_ptr() as *const libc::c_void, arp_request.len());
    }

    // 7. Poll the interface — this should produce an ARP reply
    let timestamp = smoltcp::time::Instant::now();
    net.poll(timestamp);

    // 8. Read response from the fd
    let mut resp_buf = vec![0u8; 1514];
    let n = unsafe {
        libc::read(fd, resp_buf.as_mut_ptr() as *mut libc::c_void, resp_buf.len())
    };

    assert!(n > 0, "Expected ARP reply, got empty read (n={})", n);
    resp_buf.truncate(n as usize);

    // 9. Verify it's a valid ARP reply from netstack
    assert!(
        is_arp_reply(&resp_buf),
        "Expected ARP reply from 172.16.0.1, got {}-byte frame: {:02x?}",
        resp_buf.len(),
        &resp_buf[..resp_buf.len().min(60)]
    );

    // 10. Clean up
    guest.stop().expect("guest stop");
}

/// Build a minimal DNS A query for the given domain.
///
/// Returns the raw DNS wire-format query bytes.
fn build_dns_query(domain: &str, id: u16) -> Vec<u8> {
    let mut msg = Vec::with_capacity(512);

    // Header (12 bytes)
    msg.extend_from_slice(&id.to_be_bytes());  // ID
    msg.extend_from_slice(&[0x01, 0x00]);      // flags: RD=1
    msg.extend_from_slice(&[0x00, 0x01]);      // QDCOUNT = 1
    msg.extend_from_slice(&[0x00, 0x00]);      // ANCOUNT = 0
    msg.extend_from_slice(&[0x00, 0x00]);      // NSCOUNT = 0
    msg.extend_from_slice(&[0x00, 0x00]);      // ARCOUNT = 0

    // Question: QNAME (length-prefixed labels)
    for label in domain.split('.') {
        msg.push(label.len() as u8);
        msg.extend_from_slice(label.as_bytes());
    }
    msg.push(0x00);  // null terminator

    // QTYPE: A (1), QCLASS: IN (1)
    msg.extend_from_slice(&[0x00, 0x01]);
    msg.extend_from_slice(&[0x00, 0x01]);

    msg
}

/// Parse a minimal DNS response and return true if it has at least one A record.
fn is_valid_dns_response(data: &[u8]) -> bool {
    if data.len() < 12 {
        return false;
    }
    // Check QR flag = 1 (response)
    if data[2] & 0x80 == 0 {
        return false;
    }
    // Check RCODE = 0 (no error)
    if data[3] & 0x0f != 0 {
        return false;
    }
    // Check answer count > 0
    let ancount = u16::from_be_bytes([data[6], data[7]]);
    ancount > 0
}

#[test]
#[ignore = "requires VM entitlement + kernel + initrd artifacts"]
fn test_dns_proxy_vsock() {
    // 1. Build guest config with networking and dns_vsock_port
    let config = GuestConfig::builder()
        .kernel_path(kernel_path())
        .initrd_path(Some(initrd_path()))
        .cmdline("console=hvc0".to_string())
        .cpu_count(1)
        .memory_size_bytes(512 * 1024 * 1024)
        .vsock_port(1234)                           // control vsock
        .dns_vsock_port(1235)                       // DNS proxy vsock (D-07)
        .network(speck_vz::NetworkConfig {
            subnet: std::net::Ipv4Addr::new(172, 16, 0, 0),
            gateway: std::net::Ipv4Addr::new(172, 16, 0, 1),
            guest_ip: std::net::Ipv4Addr::new(172, 16, 0, 2),
            netmask: 24,
            mtu: 1500,
            mac_address: [0x02, 0x00, 0x00, 0x00, 0x00, 0x02],
        })
        .stop_timeout(Duration::from_secs(10))
        .build();

    // 2. Create and start the guest
    let mut guest = Guest::new().expect("Guest::new");
    guest.start(&config).expect("guest start");

    // 3. Connect to guest DNS proxy vsock port 1235
    //    This returns a VzSocket whose fd the host DNS proxy reads from
    let vsock = guest.vsock_connect(1235)
        .expect("vsock_connect to DNS proxy port");

    // 4. Build and send a length-prefixed DNS A query for "google.com"
    let query = build_dns_query("google.com", 0x1234);
    let query_len = query.len() as u16;

    // Write length prefix (u16 BE) + query bytes
    let mut wire = Vec::with_capacity(2 + query.len());
    wire.extend_from_slice(&query_len.to_be_bytes());
    wire.extend_from_slice(&query);

    unsafe {
        libc::write(vsock.as_raw_fd(), wire.as_ptr() as *const libc::c_void, wire.len());
    }

    // 5. Read response: u16 BE length prefix + DNS response bytes
    let mut len_buf = [0u8; 2];
    let n = unsafe {
        libc::read(vsock.as_raw_fd(), len_buf.as_mut_ptr() as *mut libc::c_void, 2)
    };
    assert_eq!(n, 2, "should read 2-byte length prefix");

    let resp_len = u16::from_be_bytes(len_buf) as usize;
    assert!(resp_len > 0 && resp_len <= 512,
        "DNS response length {} out of range", resp_len);

    let mut resp_buf = vec![0u8; resp_len];
    let n = unsafe {
        libc::read(vsock.as_raw_fd(), resp_buf.as_mut_ptr() as *mut libc::c_void, resp_len)
    };
    assert_eq!(n as usize, resp_len,
        "should read {} DNS response bytes, got {}", resp_len, n);

    // 6. Verify it's a valid DNS response
    assert!(
        is_valid_dns_response(&resp_buf),
        "expected valid DNS response A record for google.com, got {} bytes: {:02x?}",
        resp_buf.len(),
        &resp_buf[..resp_buf.len().min(40)]
    );

    // 7. Clean up
    guest.stop().expect("guest stop");
}
